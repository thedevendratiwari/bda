# Paste:

from airflow import DAG
from airflow.providers.standard.operators.bash import BashOperator
from datetime import datetime


with DAG(
    dag_id="telecom_customer_usage_etl",
    start_date=datetime(2026, 10, 1),
    schedule=None,
    catchup=False,
) as dag:

    # 1. EXTRACT
    extract = BashOperator(
        task_id="extract",
        bash_command="""
        test -f /home/ubuntu/telecom-usage-etl/data/telecom_usage.csv
        echo "Input CSV file found successfully"
        """,
    )

    # 2. LOAD RAW DATA INTO HDFS
    load_to_hdfs = BashOperator(
        task_id="load_to_hdfs",
        bash_command="""
        hdfs dfs -mkdir -p /telecom/raw/customer_usage

        hdfs dfs -put -f \
        /home/ubuntu/telecom-usage-etl/data/telecom_usage.csv \
        /telecom/raw/customer_usage/

        echo "CSV loaded into HDFS successfully"

        hdfs dfs -ls /telecom/raw/customer_usage
        """,
    )

    # 3. TRANSFORM DATA USING HIVE
    transform_hive = BashOperator(
        task_id="transform_hive",
        bash_command="""
        beeline -u jdbc:hive2://localhost:10000/default -e "

        CREATE DATABASE IF NOT EXISTS telecom_dw;

        USE telecom_dw;

        DROP TABLE IF EXISTS telecom_usage_raw;

        CREATE TABLE telecom_usage_raw (
            usage_id STRING,
            customer_id STRING,
            plan STRING,
            call_minutes INT,
            sms_count INT,
            data_gb DOUBLE,
            usage_date DATE,
            circle STRING
        )
        ROW FORMAT DELIMITED
        FIELDS TERMINATED BY ','
        STORED AS TEXTFILE
        TBLPROPERTIES ('skip.header.line.count'='1');

        LOAD DATA INPATH
        '/telecom/raw/customer_usage/telecom_usage.csv'
        INTO TABLE telecom_usage_raw;

        DROP TABLE IF EXISTS telecom_usage_final;

        CREATE TABLE telecom_usage_final AS
        SELECT
            usage_id,
            customer_id,
            plan,
            call_minutes,
            sms_count,
            data_gb,
            usage_date,
            circle,
            CASE
                WHEN data_gb >= 10 THEN 'HIGH'
                ELSE 'NORMAL'
            END AS usage_category
        FROM telecom_usage_raw;
        "
        """,
    )

    # 4. LOAD DATA WAREHOUSE / VERIFY FINAL TABLE
    load_dw = BashOperator(
        task_id="load_dw",
        bash_command="""
        beeline -u jdbc:hive2://localhost:10000/default -e "

        USE telecom_dw;

        SELECT COUNT(*) AS raw_count
        FROM telecom_usage_raw;

        SELECT COUNT(*) AS final_count
        FROM telecom_usage_final;

        SELECT
            usage_id,
            customer_id,
            plan,
            data_gb,
            circle,
            usage_category
        FROM telecom_usage_final;

        SELECT
            usage_category,
            COUNT(*) AS count
        FROM telecom_usage_final
        GROUP BY usage_category;
        "
        """,
    )


    # DAG DEPENDENCY
    extract >> load_to_hdfs >> transform_hive >> load_dw

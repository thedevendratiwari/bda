//Problem Statement

An online learning platform stores information about learners, courses, and course enrollments.

The data engineering team needs to build a Hive-based analytics system to answer questions such as:

Which learners are enrolled in which courses?
Which courses have the highest number of enrollments?
Which learners have not enrolled in any course?
Which courses currently have no learners?
How can small reference tables such as course information be joined efficiently with a large enrollment table?
How can enrollment data be partitioned so that queries process only the required data?

The organization wants to store raw data in HDFS and use Hive for SQL-based analytics.

SELECT * FROM enrollments;


==================================================
# 16. INNER JOIN
==================================================
SET hive.auto.convert.join=false;

SELECT
    l.learner_id,
    l.learner_name,
    c.course_name
FROM learners l
INNER JOIN enrollments e
ON l.learner_id = e.learner_id
INNER JOIN courses c
ON e.course_id = c.course_id;


==================================================
# 17. LEFT OUTER JOIN
# Learners with / without courses
==================================================

SELECT
    l.learner_id,
    l.learner_name,
    e.course_id
FROM learners l
LEFT OUTER JOIN enrollments e
ON l.learner_id = e.learner_id;


==================================================
# 18. RIGHT OUTER JOIN
==================================================

SELECT
    l.learner_id,
    l.learner_name,
    e.course_id
FROM learners l
RIGHT OUTER JOIN enrollments e
ON l.learner_id = e.learner_id;


==================================================
# 19. FULL OUTER JOIN
==================================================

SELECT
    l.learner_id,
    l.learner_name,
    e.course_id
FROM learners l
FULL OUTER JOIN enrollments e
ON l.learner_id = e.learner_id;


==================================================
# 20. LEARNERS WITH NO ENROLLMENT
==================================================

SELECT
    l.learner_id,
    l.learner_name
FROM learners l
LEFT JOIN enrollments e
ON l.learner_id = e.learner_id
WHERE e.learner_id IS NULL;


==================================================
# 21. COURSES WITH NO LEARNERS
==================================================

SELECT
    c.course_id,
    c.course_name
FROM courses c
LEFT JOIN enrollments e
ON c.course_id = e.course_id
WHERE e.course_id IS NULL;


==================================================
# 22. COURSES WITH HIGHEST ENROLLMENTS
==================================================

SELECT
    c.course_id,
    c.course_name,
    COUNT(e.enrollment_id) AS enrollment_count
FROM courses c
JOIN enrollments e
ON c.course_id = e.course_id
GROUP BY c.course_id, c.course_name
ORDER BY enrollment_count DESC;


==================================================
# 23. MAP-SIDE JOIN
==================================================

# Small table = courses
# Large table = enrollments

//Alternatively, force it with a hint (works regardless of the auto-convert setting in most versions):

SELECT /*+ MAPJOIN(c) */
    e.enrollment_id,
    e.learner_id,
    c.course_name
FROM enrollments e
JOIN courses c
ON e.course_id = c.course_id;


==================================================
# 24. REDUCE-SIDE JOIN
==================================================



SET hive.auto.convert.join=false;

SELECT
    e.enrollment_id,
    e.learner_id,
    c.course_name
FROM enrollments e
JOIN courses c
ON e.course_id = c.course_id;


==================================================
# 25. PARTITIONED TABLE
==================================================

CREATE TABLE enrollments_partitioned (
    enrollment_id STRING,
    learner_id STRING,
    course_id STRING
)
PARTITIONED BY (
    enrollment_date DATE
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ',';


==================================================
# 26. INSERT INTO PARTITIONS
==================================================

INSERT INTO TABLE enrollments_partitioned
PARTITION (enrollment_date='2026-10-01')
SELECT
    enrollment_id,
    learner_id,
    course_id
FROM enrollments
WHERE enrollment_date='2026-10-01';


INSERT INTO TABLE enrollments_partitioned
PARTITION (enrollment_date='2026-10-02')
SELECT
    enrollment_id,
    learner_id,
    course_id
FROM enrollments
WHERE enrollment_date='2026-10-02';


INSERT INTO TABLE enrollments_partitioned
PARTITION (enrollment_date='2026-10-03')
SELECT
    enrollment_id,
    learner_id,
    course_id
FROM enrollments
WHERE enrollment_date='2026-10-03';


==================================================
# 27. SHOW PARTITIONS
==================================================

SHOW PARTITIONS enrollments_partitioned;


==================================================
# 28. QUERY PARTITION
==================================================

SELECT *
FROM enrollments_partitioned
WHERE enrollment_date='2026-10-02';


==================================================
# 29. PARTITION + JOIN
==================================================

SELECT
    e.enrollment_id,
    e.learner_id,
    c.course_name
FROM enrollments_partitioned e
JOIN courses c
ON e.course_id = c.course_id
WHERE e.enrollment_date='2026-10-02';
